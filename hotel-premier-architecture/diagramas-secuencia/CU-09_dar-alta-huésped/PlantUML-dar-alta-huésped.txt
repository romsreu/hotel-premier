actor Usuario
participant "UI_AltaHuesped" as UI
participant "GestorHuesped" as Gestor
participant "HuespedDAO" as HuespedDAO
participant "DireccionDAO" as DireccionDAO

Usuario -> UI : Solicita dar alta de huésped

UI -> Usuario : Muestra formulario con campos

loop Hasta que los datos sean válidos
    Usuario -> UI : Completa campos del formulario
    
    alt Datos incompletos o inválidos
        UI -> Usuario : Muestra errores de validación
    end
end

Usuario -> UI : Presiona "Siguiente"

UI -> Gestor : cargar(DarAltaHuespedDTO)

Gestor -> HuespedDAO : existePorDocumento(numeroDoc, tipoDoc)
HuespedDAO --> Gestor : boolean

alt Huésped ya existe
    Gestor --> UI : Exception: "Ya existe huésped con ese documento"
    UI -> Usuario : Muestra advertencia "DNI duplicado"
else Huésped no existe
    Gestor -> DireccionDAO : buscarPorDatos(direccion)
    DireccionDAO --> Gestor : Direccion o null
    
    alt Dirección no existe
        Gestor -> DireccionDAO : guardar(Direccion)
        DireccionDAO --> Gestor : Direccion
    end
    
    Gestor -> HuespedDAO : guardar(Huesped)
    HuespedDAO --> Gestor : Huesped
    
    Gestor --> UI : Huesped
    UI -> Usuario : Muestra confirmación de éxito
    UI -> Usuario : Vuelve al menú principal
end

alt Usuario presiona "Cancelar"
    Usuario -> UI : Presiona "Cancelar"
    UI -> Usuario : Vuelve al menú principal
end
