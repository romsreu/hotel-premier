actor Usuario
participant "UI_EstadoHabs1" as UI1
participant "UI_EstadoHabs2" as UI2
participant "GestorHabitacion" as Gestor
participant "HabitacionDAO" as HabDAO
participant "EstadoHabitacionDAO" as EstadoDAO

Usuario -> UI1 : Solicita ver estado de habitaciones

UI1 -> Usuario : Muestra campos de fechas (desde/hasta)

Usuario -> UI1 : Ingresa fechas y presiona "Mostrar"

UI1 -> UI2 : Navega con fechas seleccionadas

UI2 -> Gestor : listarTodasHabitaciones()
Gestor -> HabDAO : listarTodas()
HabDAO --> Gestor : List<Habitacion>
Gestor --> UI2 : List<Habitacion>

UI2 -> Gestor : obtenerEstadosEnRango(numeros, fechaInicio, fechaFin)
Gestor -> EstadoDAO : obtenerEstadosEnRango(numeros, fechaInicio, fechaFin)
EstadoDAO --> Gestor : Map<String, EstadoHab>
Gestor --> UI2 : Map<String, EstadoHab>

UI2 -> Usuario : Muestra grilla con estados coloreados

alt Usuario hace clic en celda
  Usuario -> UI2 : Click en celda
  UI2 -> Usuario : Muestra información de la habitación
end

alt Usuario presiona "Volver"
  Usuario -> UI2 : Presiona "Volver"
  UI2 -> UI1 : Vuelve a selección de fechas
end

alt Usuario presiona "Cancelar"
  Usuario -> UI2 : Presiona "Cancelar"
  UI2 -> Usuario : Vuelve al menú principal
end
